# WorkflowExecutionCanceledResponse

Emitted when a workflow execution is canceled.

This is a terminal event indicating the workflow was explicitly canceled.

## Example Usage

```typescript
import { WorkflowExecutionCanceledResponse } from "@mistralai/mistralai/models/components";

let value: WorkflowExecutionCanceledResponse = {
  eventId: "<id>",
  eventTimestamp: 273716,
  rootWorkflowExecId: "<id>",
  parentWorkflowExecId: "<id>",
  continuedRunId: null,
  firstExecutionRunId: "<id>",
  scheduleId: null,
  workflowExecId: "<id>",
  workflowRunId: "<id>",
  workflowName: "<value>",
  attributes: {
    taskId: "<id>",
  },
};
```
